# Local development + full test suite (pytest, ruff, black, MCP e2e).
# Production / DO deploy uses requirements.txt ([mcp] only).
# mysql extra: the startup-budget test imports PyMySQL. Without it the
# constructor raises "install PyMySQL" and never reaches the budget error.
-e ".[dev,mcp,mysql]"

# ML stack pins — apply AFTER editable install (pyproject allows wide ranges).
# Unpinned `sentence-transformers>=2.2.0` currently resolves to 5.6.0, which pulls
# transformers 5.x while torch stays at 2.2.x → ~62 embedding tests error on model load.
# numpy>=2 can break the torch 2.2 / scipy stack used in tests.
transformers==4.40.2
sentence-transformers==2.7.0
numpy<2
# black>=24 floated to 26.x and changed paren wrapping, so `black --check`
# failed before pytest. Pin the version the tree is formatted to.
black==26.5.1
# mcp 2.x removed FastMCP (mcp.server.fastmcp). pyproject already says <2;
# re-assert it here so a later unconstrained install cannot stay on 2.x.
mcp>=1.13,<2

# Third-party corpus adapters (MT-Bench parquet on Hugging Face)
pyarrow>=15.0.0

# EU AI Act offline compliance gate (scripts/compliance/check.sh + CI)
trustlint>=2.0.1
